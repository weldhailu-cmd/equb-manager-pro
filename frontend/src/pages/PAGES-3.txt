// 2. ADMIN DASHBOARD COMPONENT
function AdminDashboard() {
  const navigate = useNavigate();
  const [darkMode, setDarkMode] = useState(true);

  const [equbInfo] = useState({
    name: "Yop Digital Equb",
    city: "አዲስ አበባ",
    contributionPerSlot: 1000,
    bankName: "ፀደይ ባንክ (Tsedey Bank)",
    accountNumber: "100045896214",
    serviceFeeRate: 0.06,
  });

  const [members, setMembers] = useState([
    { id: 1, name: "አበበ ከበደ", phone: "+251911223344", slots: 2, status: "Paid", winner: false },
    { id: 2, name: "መከተ አሰፋ", phone: "+251922334455", slots: 1, status: "Paid", winner: false },
    { id: 3, name: "ሰላማዊት ታደሰ", phone: "+251933445566", slots: 3, status: "Pending", winner: false },
    { id: 4, name: "ዮናታን ዳዊት", phone: "+251944556677", slots: 4, status: "Paid", winner: false },
  ]);

  const [winnerMessage, setWinnerMessage] = useState("");
  const [newSubMember, setNewSubMember] = useState({ name: "", phone: "", slots: 1 });
  const [showAddModal, setShowAddModal] = useState(false);

  const totalSlotsCount = members.reduce((acc, m) => acc + Number(m.slots), 0);
  const totalPoolAmount = totalSlotsCount * equbInfo.contributionPerSlot;
  const serviceFeeAmount = totalPoolAmount * equbInfo.serviceFeeRate;
  const netWinnerAmount = totalPoolAmount - serviceFeeAmount;

  const handleDrawWinner = () => {
    const eligibleMembers = members.filter((m) => !m.winner);
    if (eligibleMembers.length === 0) {
      setWinnerMessage("ሁሉም አባላት ዕጣ ደርሰዋቸዋል!");
      return;
    }
    const randomIndex = Math.floor(Math.random() * eligibleMembers.length);
    const selectedWinner = eligibleMembers[randomIndex];

    setMembers(members.map((m) => (m.id === selectedWinner.id ? { ...m, winner: true } : m)));
    setWinnerMessage(`🎉 እንኳን ደስ አሎት! አሸናፊ፦ ${selectedWinner.name} (${selectedWinner.phone}) — የተጣራ ድርሻ: ${netWinnerAmount.toLocaleString()} ETB (ኤስኤምኤስ ተልኳል)`);
  };

  const handleAddMemberSubmit = (e) => {
    e.preventDefault();
    if (!newSubMember.name || !newSubMember.phone) return;
    const newItem = {
      id: members.length + 1,
      name: newSubMember.name,
      phone: newSubMember.phone,
      slots: Number(newSubMember.slots),
      status: "Paid",
      winner: false,
    };
    setMembers([...members, newItem]);
    setNewSubMember({ name: "", phone: "", slots: 1 });
    setShowAddModal(false);
  };

  return (
    <div className={`min-h-screen ${darkMode ? "bg-[#090D16] text-gray-100" : "bg-gray-50 text-gray-900"} font-sans transition-colors duration-300 flex flex-col justify-between`}>
      
      {/* Dashboard Header */}
      <header className={`w-full px-6 py-4 border-b ${darkMode ? "border-gray-800 bg-[#090D16]/95" : "border-gray-200 bg-white/95"} sticky top-0 z-50 backdrop-blur-md flex justify-between items-center shadow-lg`}>
        <div className="flex items-center space-x-3 cursor-pointer" onClick={() => navigate("/")}>
          <div className="w-10 h-10 bg-gradient-to-tr from-emerald-600 to-teal-500 rounded-xl flex items-center justify-center font-bold shadow-md text-white text-lg">
            🌿
          </div>
          <div>
            <h1 className="text-base font-extrabold tracking-wide text-emerald-400">Yop Digital Equb</h1>
            <p className="text-[11px] text-amber-400 font-medium tracking-wide">የእቁብ አስተዳደር ዳሽቦርድ</p>
          </div>
        </div>

        <div className="flex items-center space-x-3">
          <button
            onClick={() => setDarkMode(!darkMode)}
            className={`p-2 rounded-xl border text-sm transition cursor-pointer ${darkMode ? "border-gray-700 bg-gray-900 text-amber-400 hover:bg-gray-800" : "border-gray-300 bg-gray-100 text-gray-700 hover:bg-gray-200"}`}
          >
            {darkMode ? "☀️" : "🌙"}
          </button>
          <button
            onClick={() => navigate("/")}
            className="bg-gray-800 hover:bg-gray-700 text-gray-200 px-4 py-2 rounded-xl font-bold text-xs transition border border-gray-700 cursor-pointer"
          >
            ← ወደ ዋናው ገጽ (Home)
          </button>
        </div>
      </header>

      {/* Main Dashboard Content */}
      <main className="max-w-7xl mx-auto px-6 py-10 flex-grow w-full space-y-8">
        
        {/* Meta Card */}
        <div className={`border rounded-3xl p-6 md:p-8 shadow-2xl flex flex-col md:flex-row justify-between items-start md:items-center gap-6 ${darkMode ? "bg-gray-900/80 border-gray-800" : "bg-white border-gray-200"}`}>
          <div>
            <div className="flex items-center gap-2 mb-2">
              <span className="bg-emerald-950 border border-emerald-600/50 text-emerald-300 px-3 py-1 rounded-full text-xs font-semibold">📍 {equbInfo.city}</span>
              <span className="bg-amber-950 border border-amber-600/50 text-amber-300 px-3 py-1 rounded-full text-xs font-semibold">🏦 {equbInfo.bankName}</span>
            </div>
            <h2 className="text-2xl md:text-3xl font-extrabold text-white">{equbInfo.name}</h2>
            <p className="text-xs text-gray-400 mt-1">አካውንት ቁጥር: <span className="text-emerald-400 font-mono font-bold">{equbInfo.accountNumber}</span></p>
          </div>

          <div className="flex flex-wrap gap-4">
            <div className={`border px-5 py-3 rounded-2xl text-center ${darkMode ? "bg-gray-950 border-gray-800" : "bg-gray-50 border-gray-200"}`}>
              <p className="text-xs text-gray-400">ጠቅላላ ስሎቶች (Slots)</p>
              <p className="text-xl font-extrabold text-emerald-400">{totalSlotsCount}</p>
            </div>
            <div className={`border px-5 py-3 rounded-2xl text-center ${darkMode ? "bg-gray-950 border-gray-800" : "bg-gray-50 border-gray-200"}`}>
              <p className="text-xs text-gray-400">የገንዘብ መጠን (Pool)</p>
              <p className="text-xl font-extrabold text-amber-400">{totalPoolAmount.toLocaleString()} ETB</p>
            </div>
          </div>
        </div>

        {/* Draw Winner */}
        <div className={`border rounded-3xl p-6 md:p-8 shadow-2xl ${darkMode ? "bg-gray-900/80 border-gray-800" : "bg-white border-gray-200"}`}>
          <div className="flex flex-col md:flex-row justify-between items-center gap-4 mb-6">
            <div>
              <h3 className="text-xl font-bold text-emerald-400">🎲 የዕጣ ማውጫ አስተዳደር (Equb Draw System)</h3>
              <p className="text-xs text-gray-400 mt-1">በዋና ስሎቶች እና አባላት ብዛት ላይ ተመስርቶ አውቶማቲክ አሸናፊ ይምረጡ።</p>
            </div>
            <button
              onClick={handleDrawWinner}
              className="bg-emerald-500 hover:bg-emerald-400 text-gray-950 px-6 py-3 rounded-xl font-bold text-sm shadow-xl transition cursor-pointer"
            >
              ዕጣ አውጣ (Draw Winner)
            </button>
          </div>

          {winnerMessage && (
            <div className="p-4 rounded-2xl bg-emerald-500/20 border border-emerald-500 text-emerald-200 font-medium text-sm text-center mb-6">
              {winnerMessage}
            </div>
          )}

          <div className="grid grid-cols-1 md:grid-cols-3 gap-4 text-xs text-gray-400">
            <div className={`p-4 rounded-2xl border ${darkMode ? "bg-gray-950 border-gray-800" : "bg-gray-50 border-gray-200"}`}>
              <span>የአገልግሎት ክፍያ (6%):</span>
              <p className="text-base font-bold text-red-400 mt-1">{serviceFeeAmount.toLocaleString()} ETB</p>
            </div>
            <div className={`p-4 rounded-2xl border ${darkMode ? "bg-gray-950 border-gray-800" : "bg-gray-50 border-gray-200"}`}>
              <span>ለአሸናፊው የሚሰጥ የተጣራ ገንዘብ:</span>
              <p className="text-base font-bold text-emerald-400 mt-1">{netWinnerAmount.toLocaleString()} ETB</p>
            </div>
            <div className={`p-4 rounded-2xl border ${darkMode ? "bg-gray-950 border-gray-800" : "bg-gray-50 border-gray-200"}`}>
              <span>አውቶማቲክ ኤስኤምኤስ:</span>
              <p className="text-base font-bold text-amber-400 mt-1">ንቁ (Active)</p>
            </div>
          </div>
        </div>

        {/* Members Table */}
        <div className={`border rounded-3xl p-6 md:p-8 shadow-2xl ${darkMode ? "bg-gray-900/80 border-gray-800" : "bg-white border-gray-200"}`}>
          <div className="flex justify-between items-center mb-6">
            <div>
              <h3 className="text-xl font-bold text-white">👥 የአባላት ዝርዝር (Members & Sub-slots)</h3>
            </div>
            <button
              onClick={() => setShowAddModal(true)}
              className="px-4 py-2 rounded-xl border border-emerald-600/50 bg-emerald-950 text-emerald-300 text-xs font-semibold hover:bg-emerald-900 transition cursor-pointer"
            >
              + አባል ጨምር
            </button>
          </div>

          <div className="overflow-x-auto">
            <table className="w-full text-left border-collapse">
              <thead>
                <tr className={`border-b text-xs uppercase ${darkMode ? "border-gray-800 text-gray-400" : "border-gray-200 text-gray-600"}`}>
                  <th className="py-3 px-4">ተ.ቁ</th>
                  <th className="py-3 px-4">የአባል ስም</th>
                  <th className="py-3 px-4">ስልክ ቁጥር</th>
                  <th className="py-3 px-4">ስሎቶች</th>
                  <th className="py-3 px-4">ሁኔታ</th>
                  <th className="py-3 px-4">ዕጣ ውጤት</th>
                </tr>

                        {m.status}
                      </span>
                    </td>
                    <td className="py-4 px-4">
                      {m.winner ? (
                        <span className="bg-amber-400 text-gray-950 font-bold px-3 py-1 rounded-full text-xs">🏆 አሸናፊ</span>

                      )}
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        </div>

      </main>

      {/* Add Modal */}
      {showAddModal && (
        <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/75 backdrop-blur-sm p-4">

                <label className="block text-xs font-semibold text-gray-400 mb-1">ስልክ ቁጥር (Phone)</label>
                <input
                  type="text"
                  required
                  placeholder="+2519..."
                  value={newSubMember.phone}
                  onChange={(e) => setNewSubMember({ ...newSubMember, phone: e.target.value })}
                  className={`w-full px-4 py-3 rounded-xl border text-sm outline-none ${darkMode ? "bg-gray-950 border-gray-700 text-white" : "bg-gray-50 border-gray-300"}`}
                />
              </div>
              <div>
                <label className="block text-xs font-semibold text-gray-400 mb-1">የስሎቶች ብዛት (Slots)</label>
                <input
                  type="number"
                  min="1"
                  max="10"
                  required
                  value={newSubMember.slots}
                  onChange={(e) => setNewSubMember({ ...newSubMember, slots: e.target.value })}
                  className={`w-full px-4 py-3 rounded-xl border text-sm outline-none ${darkMode ? "bg-gray-950 border-gray-700 text-white" : "bg-gray-50 border-gray-300"}`}
                />
              </div>
              <div className="flex gap-3 mt-6">
                <button

                  type="submit"

                >
                  መዝግብ
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* Footer */}

  );
